# Bắt đầu nhanh Voice SFU

Mỗi ví dụ dưới đây kết nối vào phòng thoại Mezon (`CHANNEL_TYPE_MEZON_VOICE = 10`) bằng `sfuUrl` và JWT HS256 `token` lấy từ [`GenerateMeetToken`](./concepts-and-tokens.md#xin-meet-token-generatemeettoken). Lưu ý rằng mọi phiên SFU đều bắt đầu ở trạng thái **tắt mic phía máy chủ** — hãy bật micro (hoặc push-to-talk với vai trò audience) trước khi gửi âm thanh.

:::warning Các SDK ở trạng thái Pre-1.0
Cả sáu SFU SDK đều đang ở giai đoạn pre-1.0. Với bot chạy thực tế, hãy thay hàm token tĩnh (`TokenProvider.fixed` / `sfu.StaticToken`) bằng callback `TokenProvider` gọi lại `GenerateMeetToken` mỗi khi SDK yêu cầu làm mới (`reconnect`, `expiring`, `rejected`).
:::

```ts
import wrtc from "@roamhq/wrtc";
import WebSocket from "ws";
import { Room } from "mezon-sfu-sdk";
import { createWrtcPlatform, PcmAudioSource, PcmAudioSink } from "mezon-sfu-sdk/node";

const room = new Room({
  platform: createWrtcPlatform({ wrtc, WebSocket }),
});

room.on("trackSubscribed", (track, participant) => {
  if (track.kind === "audio") {
    const sink = new PcmAudioSink(wrtc, track.mediaStreamTrack);
    sink.on("data", (frame) => {
      // Khung PCM đã giải mã từ participant (đọc sampleRate/channelCount trên mỗi frame)
      void participant;
      void frame;
    });
  }
});

const source = new PcmAudioSource(wrtc); // PCM 16-bit, mặc định 48 kHz mono
await room.connect(process.env.MEZON_SFU_URL!, process.env.MEZON_SFU_TOKEN!);
await room.localParticipant.setMicrophoneEnabled(true, { track: source.track });
```

```go
package main

import (
	"context"
	"log"
	"os"
	"time"

	sfu "github.com/mezonai/mezon-sfu-sdk-go"
	"github.com/pion/webrtc/v4/pkg/media"
)

func joinVoice(ctx context.Context, opusFrame []byte) error {
	room := sfu.NewRoom(sfu.Options{
		Handlers: sfu.Handlers{
			OnTrackSubscribed: func(track *sfu.RemoteTrack, p *sfu.RemoteParticipant) {
				go func() {
					for {
						pkt, err := track.ReadRTP() // Opus hoặc VP8 RTP; io.EOF khi hủy đăng ký
						if err != nil {
							return
						}
						_ = pkt
					}
				}()
			},
			OnMessageReceived: func(text string, from sfu.MessageSender) {
				log.Println(from.UserID, text)
			},
		},
	})

	if err := room.Connect(ctx, os.Getenv("MEZON_SFU_URL"), sfu.StaticToken(os.Getenv("MEZON_SFU_TOKEN"))); err != nil {
		return err
	}
	defer room.Disconnect()

	if err := room.LocalParticipant().SetMicrophoneEnabled(ctx, true); err != nil {
		return err
	}

	// Kèm mức âm lượng RFC 6464 (0 to nhất..127 im lặng) cho VAD chế độ fixed-slot
	return room.LocalParticipant().Microphone().WriteSampleWithLevel(
		media.Sample{Data: opusFrame, Duration: 20 * time.Millisecond},
		30,
	)
}
```

```rust
use std::time::Duration;
use mezon_sfu::{Options, Room, RoomEvent, TokenProvider};

async fn run(url: &str, token: &str, opus_frame: bytes::Bytes) -> mezon_sfu::Result<()> {
    let (room, mut events) = Room::new(Options::default());
    room.connect(url, TokenProvider::fixed(token)).await?;

    room.local_participant().set_microphone_enabled(true).await?;
    room.local_participant()
        .microphone()
        .write_opus(opus_frame, Duration::from_millis(20), Some(30))
        .await?;

    while let Some(event) = events.recv().await {
        if let RoomEvent::TrackSubscribed(track, participant) = event {
            tokio::spawn(async move {
                while let Some(packet) = track.read_rtp().await {
                    let _ = (&participant, packet);
                }
            });
        }
    }

    room.disconnect().await;
    Ok(())
}
```

```python
from mezon_sfu import Options, RemoteTrack, Room, TokenProvider, TrackSubscribed


async def run(url: str, token: str, pcm: bytes) -> None:
    async with asyncio.TaskGroup() as listeners, Room(Options()) as room:
        await room.connect(url, TokenProvider.fixed(token))
        await room.local_participant.set_microphone_enabled(True)
        room.local_participant.microphone.write_pcm(pcm, sample_rate=48_000, channels=1)

        async for event in room.events():
            if isinstance(event, TrackSubscribed) and event.track.kind == "audio":
                listeners.create_task(listen(event.track))

    async for frame in track.audio_frames():
        # PCM 48 kHz stereo 16-bit đã giải mã (khung 20 ms)
        _ = frame
```

## Mô hình Audience & Push-to-Talk

Để tham gia ở chế độ người nghe và chỉ phát âm thanh khi cần:

- **TypeScript:** `new Room({ role: "audience" })` → `await room.localParticipant.setPushToTalk(true)`
- **Go:** `sfu.NewRoom(sfu.Options{Role: sfu.RoleAudience})` → `room.LocalParticipant().SetPushToTalk(ctx, true)`
- **Rust:** `Options { role: Role::Audience, ..Default::default() }` → `room.local_participant().set_push_to_talk(true).await?`
- **Kotlin:** `Options(role = Role.AUDIENCE)` → `room.localParticipant.setPushToTalk(true)`
- **Python:** `Options(role=Role.AUDIENCE)` → `await room.local_participant.set_push_to_talk(True)`
- **.NET:** `new RoomOptions { Role = Role.Audience }` → `await room.LocalParticipant.SetPushToTalkAsync(true)`
